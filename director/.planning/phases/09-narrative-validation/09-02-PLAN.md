Handler functions:
```typescript
async function handleValidate() {
  if (!selectedAssetSetId) return;
  await runValidation('default-project', selectedAssetSetId);
  setShowValidationPanel(true);
}

async function handleRegenerateNode(nodeId: string) {
  // Reset node status to allow regeneration (NAR-03)
  // The generate-video route requires status 'pending' or 'ready'
  // Update local node status to 'pending' first, then trigger generation
  const { generateSingleVideo } = usePipelineNodeStore.getState();

  // Mark the validation issue as 'regenerated'
  const issuesForNode = validationIssues.filter(i => i.nodeId === nodeId && i.status === 'flagged');
  const { markRegenerated } = useValidationStore.getState();
  for (const issue of issuesForNode) {
    await markRegenerated(issue.id);
  }

  // Reset node to pending via direct Supabase update through a lightweight API call,
  // then trigger single video generation
  await fetch(`/api/pipeline-nodes?id=${encodeURIComponent(nodeId)}`, {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ status: 'pending' }),
  });
  // Update local state
  usePipelineNodeStore.getState().updateNodeStatus(nodeId, 'pending');
  await generateSingleVideo(nodeId);
}

async function handleAcceptSequence() {
  if (!selectedAssetSetId) return;
  await acceptAll('default-project', selectedAssetSetId);
}
```
