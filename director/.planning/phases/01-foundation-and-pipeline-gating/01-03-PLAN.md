1. Create `src/lib/pipeline/orchestrator.ts`:

```typescript
import { usePipelineStore } from '@/lib/stores/pipeline-store';
import { StageId, STAGE_ORDER } from '@/lib/types/pipeline';

export class PipelineOrchestrator {
  /**
   * Attempts to advance the pipeline past the current stage.
   *
   * Behavior:
   * - Completes the current stage
   * - If the NEXT stage has awaitUserAdvance=true, stops there (sets it active but does not auto-complete)
   * - If the NEXT stage has awaitUserAdvance=false AND canExecuteStage is true, recursively advances
   * - Returns the StageId where the pipeline stopped
   */
  advanceToNext(): StageId | null {
    const store = usePipelineStore.getState();
    const { currentRun } = store;
    if (!currentRun) return null;

    const currentIndex = STAGE_ORDER.indexOf(currentRun.currentStageId);
    if (currentIndex === -1) return null;

    // Complete current stage
    store.completeStage(currentRun.currentStageId);

    // Check if there's a next stage
    const nextIndex = currentIndex + 1;
    if (nextIndex >= STAGE_ORDER.length) {
      return null; // Pipeline complete
    }

    const nextStageId = STAGE_ORDER[nextIndex];

    // Check if we can execute the next stage
    const freshStore = usePipelineStore.getState();
    if (!freshStore.canExecuteStage(nextStageId)) {
      return currentRun.currentStageId; // Blocked, stay put
    }

    // Jump to next stage
    freshStore.jumpToStage(nextStageId);

    // Check if next stage requires user advance
    const nextStage = freshStore.getStage(nextStageId);
    if (nextStage?.awaitUserAdvance) {
      return nextStageId; // Pause here for user
    }

    // Auto-advance continues (non-recursive -- use a while loop instead for safety)
    // Actually: this should NOT auto-complete stages. The orchestrator advances
    // the pointer, but stages are completed by user action (clicking Continue).
    // The auto-advance only skips past canSkip stages that don't need user input.
    return nextStageId;
  }

  /**
   * Manually advances past a stage boundary that has awaitUserAdvance=true.
   * Called when the user explicitly confirms they're ready to proceed.
   * Returns the StageId where pipeline stopped next, or null if complete.
   */
  advanceUserBoundary(): StageId | null {
    const store = usePipelineStore.getState();
    const { currentRun } = store;
    if (!currentRun) return null;

    const currentStage = store.getStage(currentRun.currentStageId);
    if (!currentStage?.awaitUserAdvance) return null; // Not at a user boundary

    return this.advanceToNext();
  }

  /**
   * Returns true if the pipeline is currently paused at a user-advance boundary.
   */
  isPausedAtBoundary(): boolean {
    const store = usePipelineStore.getState();
    const { currentRun } = store;
    if (!currentRun) return false;

    const stage = store.getStage(currentRun.currentStageId);
    return stage?.awaitUserAdvance === true && stage.status === 'active';
  }

  /**
   * Returns the current stage ID or null if no active run.
   */
  getCurrentStageId(): StageId | null {
    const store = usePipelineStore.getState();
    return store.currentRun?.currentStageId ?? null;
  }
}
```

NOTE: The orchestrator uses `usePipelineStore.getState()` (not the hook) because it's a plain class, not a React component. This is the standard Zustand pattern for accessing state outside React.

```typescript
import { useMemo } from 'react';
import { usePipelineStore } from '@/lib/stores/pipeline-store';
import { PipelineOrchestrator } from '@/lib/pipeline/orchestrator';
import { StageId } from '@/lib/types/pipeline';

interface OrchestratorHook {
  /** Advance past current stage (auto-skips non-awaitUserAdvance stages) */
  advanceToNext: () => StageId | null;
  /** Manually advance past an awaitUserAdvance boundary */
  advanceUserBoundary: () => StageId | null;
  /** Whether pipeline is paused at a user-advance boundary */
  isPausedAtBoundary: boolean;
  /** Current stage ID */
  currentStageId: StageId | null;
  /** Whether there is an active pipeline run */
  isActive: boolean;
  /** Start a new pipeline run */
  startRun: () => void;
  /** Reset to freestyle mode */
  resetRun: () => void;
}

export function usePipelineOrchestrator(): OrchestratorHook {
  const orchestrator = useMemo(() => new PipelineOrchestrator(), []);
  const currentRun = usePipelineStore((s) => s.currentRun);
  const startRun = usePipelineStore((s) => s.startRun);
  const resetRun = usePipelineStore((s) => s.resetRun);

  return {
    advanceToNext: () => orchestrator.advanceToNext(),
    advanceUserBoundary: () => orchestrator.advanceUserBoundary(),
    isPausedAtBoundary: orchestrator.isPausedAtBoundary(),
    currentStageId: currentRun?.currentStageId ?? null,
    isActive: currentRun !== null,
    startRun,
    resetRun,
  };
}
```

This hook:
- Creates a stable orchestrator instance via useMemo
- Subscribes to reactive store state (currentRun) so components re-render when pipeline state changes
- Exposes a clean API for components: advanceToNext, advanceUserBoundary, isPausedAtBoundary, startRun, resetRun
- Does NOT require components to know about PipelineOrchestrator class
