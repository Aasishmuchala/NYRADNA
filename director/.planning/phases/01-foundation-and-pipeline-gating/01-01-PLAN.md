```typescript
import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';
import { StageId, PipelineRun, PipelineStage, STAGE_ORDER, DEFAULT_STAGES } from '../types/pipeline';

interface PipelineState {
  currentRun: PipelineRun | null;

  // Actions
  startRun: () => void;
  resetRun: () => void;
  completeStage: (stageId: StageId) => void;
  jumpToStage: (stageId: StageId) => boolean;

  // Queries (pure functions derived from state)
  canExecuteStage: (stageId: StageId) => boolean;
  getStage: (stageId: StageId) => PipelineStage | undefined;
  getStageStatus: (stageId: StageId) => StageStatus;
}
```

Implement the gating logic:

**canExecuteStage(stageId):**
- If `currentRun` is null, return `true` (freestyle mode -- GATE-05)
- Find the target stage's index in STAGE_ORDER
- For every stage BEFORE the target in STAGE_ORDER: if that stage has `canSkip === false` AND `status !== 'completed'`, return `false`
- Otherwise return `true`

**jumpToStage(stageId):**
- Call `canExecuteStage(stageId)` -- if false, return false (GATE-02)
- Set `currentStageId` to the target
- Update the target stage status to 'active'
- Return true

**completeStage(stageId):**
- Mark stage status as 'completed'
- Determine next stage in STAGE_ORDER
- If next stage exists, set it as currentStageId and mark 'active'

**startRun():**
- Generate a run ID using `crypto.randomUUID()`
- Create PipelineRun with cloned DEFAULT_STAGES (all idle), currentStageId = 'intent'
- Set first stage to 'active'

**resetRun():**
- Set `currentRun` to null

Use Zustand persist middleware with `sessionStorage` for persistence:
```typescript
export const usePipelineStore = create<PipelineState>()(
  persist(
    (set, get) => ({ ... }),
    {
      name: 'pipeline-store',
      storage: createJSONStorage(() => sessionStorage),
    }
  )
);
```
