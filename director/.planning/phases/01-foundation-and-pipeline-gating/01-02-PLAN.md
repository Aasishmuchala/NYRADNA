1. Create `src/hooks/useGatedNavigation.ts`:

```typescript
import { useRouter } from 'next/navigation';
import { usePipelineStore } from '@/lib/stores/pipeline-store';
import { StageId, STAGE_ORDER } from '@/lib/types/pipeline';

interface GatedNavResult {
  canProceed: boolean;
  blockedBy: StageId | null;
  blockedByLabel: string | null;
  blockedByHref: string | null;
  errorMessage: string | null;
  navigate: () => void;
}

export function useGatedNavigation(targetStageId: StageId): GatedNavResult {
  const router = useRouter();
  const { canExecuteStage, currentRun } = usePipelineStore();

  // Freestyle mode -- no gating
  if (!currentRun) {
    const stage = currentRun === null ? null : undefined;
    // Find href from DEFAULT_STAGES
    return {
      canProceed: true,
      blockedBy: null,
      blockedByLabel: null,
      blockedByHref: null,
      errorMessage: null,
      navigate: () => router.push(getStageHref(targetStageId)),
    };
  }

  const canProceed = canExecuteStage(targetStageId);

  if (canProceed) {
    return {
      canProceed: true,
      blockedBy: null,
      blockedByLabel: null,
      blockedByHref: null,
      errorMessage: null,
      navigate: () => router.push(getStageHref(targetStageId)),
    };
  }

  // Find the first incomplete mandatory blocker
  const targetIndex = STAGE_ORDER.indexOf(targetStageId);
  const blocker = findFirstBlocker(currentRun, targetIndex);

  return {
    canProceed: false,
    blockedBy: blocker?.id ?? null,
    blockedByLabel: blocker?.label ?? null,
    blockedByHref: blocker?.href ?? null,
    errorMessage: blocker ? `Complete "${blocker.label}" first` : 'Prerequisites incomplete',
    navigate: () => {
      // Navigate to the blocking stage instead
      if (blocker?.href) router.push(blocker.href);
    },
  };
}
```

Helper function `findFirstBlocker` iterates STAGE_ORDER up to targetIndex, finds first stage where `canSkip === false` and `status !== 'completed'`. Helper `getStageHref` maps StageId to its href using DEFAULT_STAGES.
