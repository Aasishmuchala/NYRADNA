   b) If no API key (default for development), use a **deterministic mock** that generates a plausible SequencePlan from the brief data:
      ```typescript
      function generateMockSequencePlan(brief: DirectorsBrief): SequencePlan {
        // If beats exist, map each beat to a segment
        // If no beats, generate 3 default segments from storyline
        const segments: SequenceSegment[] = brief.narrativeBeats.beats.length > 0
          ? brief.narrativeBeats.beats.map((beat, i) => ({
              index: i,
              title: beat.label || `Segment ${i + 1}`,
              visualDescription: `${brief.visualStyle.mood} scene: ${beat.emotion} moment`,
              emotionalTone: beat.emotion,
              transitionFromPrevious: i === 0 ? null : beat.transition,
              durationHint: '3-5 seconds',
            }))
          : [
              { index: 0, title: 'Opening', visualDescription: `${brief.visualStyle.mood} establishing shot`, emotionalTone: 'anticipation', transitionFromPrevious: null, durationHint: '3-5 seconds' },
              { index: 1, title: 'Development', visualDescription: `${brief.storyline.theme} exploration`, emotionalTone: 'engagement', transitionFromPrevious: 'dissolve', durationHint: '5-8 seconds' },
              { index: 2, title: 'Resolution', visualDescription: `${brief.storyline.theme} payoff`, emotionalTone: 'satisfaction', transitionFromPrevious: 'fade', durationHint: '3-5 seconds' },
            ];

        return {
          id: crypto.randomUUID(),
          briefId: brief.id,
          segments,
          overallArc: `A ${brief.visualStyle.mood} narrative exploring ${brief.storyline.theme}`,
          estimatedTotalDuration: `${segments.length * 4}-${segments.length * 6} seconds`,
          createdAt: new Date().toISOString(),
        };
      }
      ```

   **Error handling:** Wrap AI call in try/catch. If AI call fails, fall back to mock. Return 500 only if mock also fails (shouldn't happen).

   After saveBrief() succeeds and BEFORE completeStage + navigate, trigger analysis:
   ```typescript
   const handleContinue = async () => {
     if (!isBriefComplete(brief)) return;
     await saveBrief();
     // Trigger analysis
     setAnalyzing(true);
     try {
       const res = await fetch('/api/briefs/analyze', {
         method: 'POST',
         headers: { 'Content-Type': 'application/json' },
         body: JSON.stringify({ brief }),
       });
       if (res.ok) {
         const data = await res.json();
         setSequencePlan(data.sequencePlan);
       }
     } catch (err) {
       console.error('Analysis failed:', err);
     } finally {
       setAnalyzing(false);
     }
     completeStage('directors-cut');
     router.push('/create/style-dna');
   };
   ```
