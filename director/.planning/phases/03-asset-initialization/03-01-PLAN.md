    ```typescript
    export type AssetType = 'image' | 'video' | 'audio';

    export interface AssetSetItem {
      id: string;
      assetSetId: string;
      name: string;
      type: AssetType;
      url: string;           // Supabase Storage public URL
      thumbnailUrl: string | null;
      metadata: Record<string, unknown>; // flexible JSONB
      position: number;      // 0-indexed ordering within set
      createdAt: string;
    }

    export interface AssetSet {
      id: string;
      projectId: string;
      name: string;
      description: string;
      items: AssetSetItem[];  // loaded eagerly for display
      createdAt: string;
      updatedAt: string;
    }

    export interface NewAssetSet {
      name: string;
      description: string;
      projectId: string;
    }

    /** Asset set is usable if it has a name and at least one item */
    export function isAssetSetComplete(set: AssetSet | null): boolean {
      if (!set) return false;
      return set.name.trim().length > 0 && set.items.length > 0;
    }
    ```

    Create `supabase/migrations/002_asset_sets.sql`:

    ```sql
    CREATE TABLE IF NOT EXISTS asset_sets (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      project_id UUID NOT NULL,
      name TEXT NOT NULL DEFAULT '',
      description TEXT NOT NULL DEFAULT '',
      created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
      updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
    );

    CREATE INDEX idx_asset_sets_project_id ON asset_sets(project_id);

    CREATE TABLE IF NOT EXISTS asset_set_items (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      asset_set_id UUID NOT NULL REFERENCES asset_sets(id) ON DELETE CASCADE,
      name TEXT NOT NULL DEFAULT '',
      type TEXT NOT NULL DEFAULT 'image' CHECK (type IN ('image', 'video', 'audio')),
      url TEXT NOT NULL DEFAULT '',
      thumbnail_url TEXT,
      metadata JSONB NOT NULL DEFAULT '{}',
      position INTEGER NOT NULL DEFAULT 0,
      created_at TIMESTAMPTZ NOT NULL DEFAULT now()
    );

    CREATE INDEX idx_asset_set_items_set_id ON asset_set_items(asset_set_id);
    CREATE INDEX idx_asset_set_items_position ON asset_set_items(asset_set_id, position);
    ```

    Follow the exact same migration naming pattern as 001_director_cuts.sql. Use snake_case for all column names. The ON DELETE CASCADE ensures items are cleaned up when a set is deleted.

    **1. Zustand store** (`src/lib/stores/asset-set-store.ts`):

    Follow the directors-brief-store pattern exactly. Use `create` from zustand with fetch-based persistence (NOT zustand/persist).

    State shape:
    ```typescript
    interface AssetSetState {
      assetSets: AssetSet[];
      activeSetId: string | null;
      loading: boolean;
      saving: boolean;

      // Queries
      getActiveSet: () => AssetSet | null;

      // Set CRUD
      loadAssetSets: (projectId: string) => Promise<void>;
      createAssetSet: (data: NewAssetSet) => Promise<AssetSet | null>;
      deleteAssetSet: (id: string) => Promise<void>;
      setActiveSet: (id: string | null) => void;

      // Item management
      addItem: (setId: string, item: Omit<AssetSetItem, 'id' | 'assetSetId' | 'createdAt' | 'position'>) => Promise<void>;
      removeItem: (setId: string, itemId: string) => Promise<void>;
      reorderItems: (setId: string, itemIds: string[]) => Promise<void>;

      // Upload
      uploadAsset: (setId: string, file: File) => Promise<void>;
    }
    ```

    Each async method:
    - Sets loading/saving
    - Calls the corresponding API route via fetch
    - Updates local state from response
    - Catches errors and logs to console (same pattern as directors-brief-store)

    `reorderItems` sends an array of item IDs in new order to the reorder endpoint, then updates local state to reflect new positions.

    `uploadAsset` creates a FormData with the file, POSTs to `/api/asset-sets/upload?setId={setId}`, receives back the created AssetSetItem, and adds it to the local set's items array.

    `getActiveSet` returns `assetSets.find(s => s.id === activeSetId) ?? null`.

    **2. Asset Sets API route** (`src/app/api/asset-sets/route.ts`):

    Copy the createServerSupabase helper pattern from briefs/route.ts exactly.

    GET: Query params `projectId`. Returns all asset_sets for that project with their items joined.
    - Fetch asset_sets where project_id = projectId, ordered by created_at desc
    - For each set, fetch asset_set_items where asset_set_id = set.id, ordered by position asc
    - Map snake_case to camelCase (same mapRow pattern as briefs route)
    - Return array of AssetSet objects with items embedded

    POST: Body `{ name, description, projectId }`. Creates a new asset_set row.
    - Validate name is non-empty
    - Insert into asset_sets, return the new row mapped to camelCase with empty items array

    **3. Items API route** (`src/app/api/asset-sets/[id]/items/route.ts`):

    Access the set ID from the route params: `params.id`
    Next.js 16 dynamic route params are accessed via: `const { id } = await params;`

    GET: Returns all items for the given asset set, ordered by position.

    POST: Body `{ name, type, url, thumbnailUrl?, metadata? }`.
    - Calculate next position: `SELECT COALESCE(MAX(position), -1) + 1 FROM asset_set_items WHERE asset_set_id = $id`
    - Insert new item with calculated position
    - Return the created item mapped to camelCase

    DELETE: Query param `itemId`.
    - Delete the item
    - Re-index remaining items' positions (UPDATE ... SET position = subquery)
    - Return 200

    **4. Reorder API route** (`src/app/api/asset-sets/[id]/items/reorder/route.ts`):

    PUT: Body `{ itemIds: string[] }`.
    - For each itemId at index i, UPDATE asset_set_items SET position = i WHERE id = itemId AND asset_set_id = $id
    - Use Promise.all for parallel updates
    - Return 200 with updated items

    **5. Upload API route** (`src/app/api/asset-sets/upload/route.ts`):

    POST: Multipart form data with `file` field, query param `setId`.
    - Extract file from FormData: `const file = formData.get('file') as File`
    - Generate storage path: `assets/${setId}/${crypto.randomUUID()}-${file.name}`
    - Upload to Supabase Storage bucket named "assets": `supabase.storage.from('assets').upload(path, buffer, { contentType: file.type })`
    - Get public URL: `supabase.storage.from('assets').getPublicUrl(path)`
    - Determine asset type from MIME: image/* -> 'image', video/* -> 'video', audio/* -> 'audio'
    - Calculate next position for the set (same as items POST)
    - Insert into asset_set_items with the storage URL
    - Return the created item mapped to camelCase

    All API routes use the same createServerSupabase pattern. All snake_case DB columns map to camelCase in responses. All errors return proper HTTP status codes with `{ error: string }` body.
